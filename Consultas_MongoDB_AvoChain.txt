AVOCHAIN — GUÍA DE CONSULTAS PARA MONGOSH
========================================

Base de datos: avochain
Colecciones: usuarios, fincas, lotes

CÓMO CONECTAR
--------------
Desde PowerShell puedes abrir mongosh así:

mongosh "mongodb://avoadmin:change-this-password@127.0.0.1:27018/avochain?authSource=admin"

Ya dentro de mongosh, selecciona la base:

use avochain

Consejos:
- Los campos de fecha son BSON Date; consulta con ISODate("AAAA-MM-DDTHH:mm:ssZ").
- Los identificadores de relaciones son ObjectId. Para buscar uno copiado de Compass usa ObjectId("...").
- Ejecuta las consultas de una en una. Puedes cambiar los límites y valores para practicar.
- Estas consultas son de lectura; no modifican ni borran datos.
- En estadoActual se usan exactamente: "Registrado", "Aprobado_Calidad", "Rechazado_En_Origen", "En Transito", "En Punto de Venta", "Vendido".

-------------------

 db.usuarios.find({ rol: "Inspector de calidad" }).pretty()
 db.usuarios.find({ rol: "Productor" }).pretty()
 db.usuarios.find({ rol: "Receptor" }).pretty()
 db.usuarios.find({ rol: "Vendedor" }).pretty()

// Mostrar solo nombre, rol y wallet, sin _id
 db.usuarios.find({}, { _id: 0, nombre: 1, rol: 1, walletAddress: 1 })

// Contar usuarios por rol
 db.usuarios.aggregate([
   { $group: { _id: "$rol", cantidad: { $sum: 1 } } },
   { $sort: { cantidad: -1 } }
 ])

// Roles con 5 o más usuarios
 db.usuarios.aggregate([
   { $group: { _id: "$rol", cantidad: { $sum: 1 } } },
   { $match: { cantidad: { $gte: 5 } } },
   { $sort: { cantidad: -1 } }
 ])

// Usuarios registrados desde el 1 de octubre de 2026
 db.usuarios.find({ fechaRegistro: { $gte: ISODate("2026-10-01T00:00:00Z") } })

3. FINCAS
---------

// Fincas activas
 db.fincas.find({ estado: "Activa" }).pretty()

// Fincas inactivas
 db.fincas.find({ estado: "Inactiva" }).pretty()

// Buscar finca por nombre
 db.fincas.find({ nombreFinca: /Palmas/i })

// Buscar por registro ICA
 db.fincas.findOne({ registroICA: "ICA-2026000-C0001" })

// Mostrar nombre, ICA y ubicación de cada finca
 db.fincas.find({}, { _id: 0, nombreFinca: 1, registroICA: 1, ubicacionGPS: 1, estado: 1 })

// Cantidad de fincas por estado
 db.fincas.aggregate([
   { $group: { _id: "$estado", cantidad: { $sum: 1 } } }
 ])

// Fincas junto con los datos de su productor
 db.fincas.aggregate([
   { $lookup: { from: "usuarios", localField: "productorId", foreignField: "_id", as: "productor" } },
   { $unwind: { path: "$productor", preserveNullAndEmptyArrays: true } },
   { $project: { _id: 0, nombreFinca: 1, registroICA: 1, estado: 1, productor: "$productor.nombre", walletProductor: "$productor.walletAddress" } },
   { $sort: { nombreFinca: 1 } }
 ])

// Fincas agrupadas por productor
 db.fincas.aggregate([
   { $lookup: { from: "usuarios", localField: "productorId", foreignField: "_id", as: "productor" } },
   { $unwind: { path: "$productor", preserveNullAndEmptyArrays: true } },
   { $group: { _id: "$productor.nombre", cantidadFincas: { $sum: 1 }, fincas: { $push: "$nombreFinca" } } },
   { $sort: { cantidadFincas: -1 } }
 ])

5. CALIDAD PRE-DESPACHO
-----------------------

// Lotes con inspección aprobada
 db.lotes.find({ "inspeccionCalidad.aprobado": true })

// Lotes rechazados en origen
 db.lotes.find({ estadoActual: "Rechazado_En_Origen", "inspeccionCalidad.aprobado": false })

// Aprobados con humedad dentro del intervalo 60–70%
 db.lotes.find({
   "inspeccionCalidad.aprobado": true,
   "inspeccionCalidad.humedad": { $gte: 60, $lte: 70 }
 })

// Inspecciones con temperatura fuera del intervalo 4–8°C
 db.lotes.find({ "inspeccionCalidad.temperatura": { $not: { $gte: 4, $lte: 8 } } })

// Inspecciones con humedad fuera del intervalo 60–70%
 db.lotes.find({ "inspeccionCalidad.humedad": { $not: { $gte: 60, $lte: 70 } } })

// Agrupar por aprobación y destino
 db.lotes.aggregate([
   { $match: { inspeccionCalidad: { $exists: true } } },
   { $group: { _id: { aprobado: "$inspeccionCalidad.aprobado", destino: "$inspeccionCalidad.destino" }, cantidad: { $sum: 1 } } },
   { $sort: { "_id.destino": 1 } }
 ])

// Promedio, mínimo y máximo de humedad por destino
 db.lotes.aggregate([
   { $match: { inspeccionCalidad: { $exists: true } } },
   { $group: { _id: "$inspeccionCalidad.destino", promedioHumedad: { $avg: "$inspeccionCalidad.humedad" }, minHumedad: { $min: "$inspeccionCalidad.humedad" }, maxHumedad: { $max: "$inspeccionCalidad.humedad" } } }
 ])

// Inspecciones junto con el nombre del inspector
 db.lotes.aggregate([
   { $match: { inspeccionCalidad: { $exists: true } } },
   { $lookup: { from: "usuarios", localField: "inspeccionCalidad.inspectorId", foreignField: "_id", as: "inspector" } },
   { $unwind: { path: "$inspector", preserveNullAndEmptyArrays: true } },
   { $project: { _id: 0, codigoLote: 1, estadoActual: 1, humedad: "$inspeccionCalidad.humedad", temperatura: "$inspeccionCalidad.temperatura", aprobado: "$inspeccionCalidad.aprobado", inspector: "$inspector.nombre" } }
 ])

-------------------------------------

// Lotes que tienen despacho
 db.lotes.find({ despacho: { $exists: true } }, { codigoLote: 1, estadoActual: 1, despacho: 1 })

// Lotes con conductor y nombre del transportista
 db.lotes.aggregate([
   { $match: { despacho: { $exists: true } } },
   { $lookup: { from: "usuarios", localField: "despacho.transportistaId", foreignField: "_id", as: "conductor" } },
   { $unwind: { path: "$conductor", preserveNullAndEmptyArrays: true } },
   { $project: { _id: 0, codigoLote: 1, estadoActual: 1, transportista: "$conductor.nombre", fechaSalida: "$despacho.fechaSalida" } }
 ])

// Lotes que reportaron al menos un incidente
 db.lotes.find({ "incidentesRuta.0": { $exists: true } }, { codigoLote: 1, estadoActual: 1, incidentesRuta: 1 })

// Incidentes separados: una fila de resultado por incidente
 db.lotes.aggregate([
   { $unwind: "$incidentesRuta" },
   { $project: { _id: 0, codigoLote: 1, estadoActual: 1, tipo: "$incidentesRuta.tipoIncidente", descripcion: "$incidentesRuta.descripcion", ubicacion: "$incidentesRuta.ubicacion", fecha: "$incidentesRuta.fechaReporte" } },
   { $sort: { fecha: -1 } }
 ])

// Contar incidentes por tipo
 db.lotes.aggregate([
   { $unwind: "$incidentesRuta" },
   { $group: { _id: "$incidentesRuta.tipoIncidente", cantidad: { $sum: 1 } } },
   { $sort: { cantidad: -1 } }
 ])

// Incidentes con coordenadas GPS registradas
 db.lotes.find({ "incidentesRuta.ubicacion.latitud": { $exists: true }, "incidentesRuta.ubicacion.longitud": { $exists: true } }, { codigoLote: 1, incidentesRuta: 1 })

// Incidentes del transportista y lote elegido: reemplaza el código
 db.lotes.find({ codigoLote: "AVO-2026-01025" }, { codigoLote: 1, incidentesRuta: 1 })

 db.lotes.find({ "recepcionDestino.temperaturaMaximaRuta": { $gt: 12 } }, { codigoLote: 1, recepcionDestino: 1 })

// Lotes recibidos con temperatura de llegada aceptable (4–10°C)
 db.lotes.find({ "recepcionDestino.temperaturaLlegada": { $gte: 4, $lte: 10 } })

 db.lotes.find({ ventaFinal: { $exists: true } }, { codigoLote: 1, estadoActual: 1, ventaFinal: 1 })

// Ventas agrupadas por día
 db.lotes.aggregate([
   { $match: { "ventaFinal.fechaVenta": { $exists: true } } },
   { $group: { _id: { $dateToString: { format: "%Y-%m-%d", date: "$ventaFinal.fechaVenta" } }, cantidadVendida: { $sum: 1 } } },
   { $sort: { _id: 1 } }
 ])

// Ventas junto con nombre del vendedor
 db.lotes.aggregate([
   { $match: { ventaFinal: { $exists: true } } },
   { $lookup: { from: "usuarios", localField: "ventaFinal.vendedorId", foreignField: "_id", as: "vendedor" } },
   { $unwind: { path: "$vendedor", preserveNullAndEmptyArrays: true } },
   { $project: { _id: 0, codigoLote: 1, fechaVenta: "$ventaFinal.fechaVenta", vendedor: "$vendedor.nombre", wallet: "$vendedor.walletAddress" } }

11. NIVEL MEDIO: AGREGACIONES DE VARIOS PASOS
--------------------------------------------

// Top 5 fincas por número de lotes y peso total producido
 db.lotes.aggregate([
   { $group: { _id: "$fincaId", cantidadLotes: { $sum: 1 }, pesoTotalKg: { $sum: "$datosProduccion.pesoKg" } } },
   { $lookup: { from: "fincas", localField: "_id", foreignField: "_id", as: "finca" } },
   { $unwind: "$finca" },
   { $project: { _id: 0, finca: "$finca.nombreFinca", cantidadLotes: 1, pesoTotalKg: 1 } },
   { $sort: { cantidadLotes: -1, pesoTotalKg: -1 } },
   { $limit: 5 }
 ])

// Promedio y cantidad de lotes por variedad y estado
 db.lotes.aggregate([
   { $group: { _id: { variedad: "$datosProduccion.variedad", estado: "$estadoActual" }, lotes: { $sum: 1 }, pesoPromedioKg: { $avg: "$datosProduccion.pesoKg" }, pesoTotalKg: { $sum: "$datosProduccion.pesoKg" } } },
   { $sort: { "_id.variedad": 1, "_id.estado": 1 } }
 ])

// Inspecciones agrupadas por inspector con nombre, cantidad aprobada/rechazada y humedad promedio
 db.lotes.aggregate([
   { $match: { inspeccionCalidad: { $exists: true } } },
   { $group: { _id: "$inspeccionCalidad.inspectorId", inspecciones: { $sum: 1 }, aprobadas: { $sum: { $cond: ["$inspeccionCalidad.aprobado", 1, 0] } }, rechazadas: { $sum: { $cond: ["$inspeccionCalidad.aprobado", 0, 1] } }, humedadPromedio: { $avg: "$inspeccionCalidad.humedad" }, temperaturaPromedio: { $avg: "$inspeccionCalidad.temperatura" } } },
   { $lookup: { from: "usuarios", localField: "_id", foreignField: "_id", as: "inspector" } },
   { $unwind: { path: "$inspector", preserveNullAndEmptyArrays: true } },
   { $project: { _id: 0, inspector: "$inspector.nombre", inspecciones: 1, aprobadas: 1, rechazadas: 1, humedadPromedio: { $round: ["$humedadPromedio", 2] }, temperaturaPromedio: { $round: ["$temperaturaPromedio", 2] } } },
   { $sort: { inspecciones: -1 } }
 ])

// Conteo de incidentes por transportista, incluyendo su nombre
 db.lotes.aggregate([
   { $unwind: "$incidentesRuta" },
   { $group: { _id: "$incidentesRuta.transportistaId", cantidadIncidentes: { $sum: 1 }, tipos: { $addToSet: "$incidentesRuta.tipoIncidente" }, lotesAfectados: { $addToSet: "$codigoLote" } } },
   { $lookup: { from: "usuarios", localField: "_id", foreignField: "_id", as: "transportista" } },
   { $unwind: { path: "$transportista", preserveNullAndEmptyArrays: true } },
   { $project: { _id: 0, transportista: "$transportista.nombre", cantidadIncidentes: 1, tipos: 1, cantidadLotesAfectados: { $size: "$lotesAfectados" } } },
   { $sort: { cantidadIncidentes: -1 } }
 ])

// Sumar cajas dañadas y promediar temperatura de llegada por receptor
 db.lotes.aggregate([
   { $match: { recepcionDestino: { $exists: true } } },
   { $group: { _id: "$recepcionDestino.receptorId", entregas: { $sum: 1 }, cajasDanadas: { $sum: "$recepcionDestino.cajasDanadas" }, temperaturaPromedio: { $avg: "$recepcionDestino.temperaturaLlegada" } } },
   { $lookup: { from: "usuarios", localField: "_id", foreignField: "_id", as: "receptor" } },
   { $unwind: { path: "$receptor", preserveNullAndEmptyArrays: true } },
   { $project: { _id: 0, receptor: "$receptor.nombre", entregas: 1, cajasDanadas: 1, temperaturaPromedio: { $round: ["$temperaturaPromedio", 2] } } },
   { $sort: { cajasDanadas: -1 } }
 ])

12. NIVEL MEDIO/ALTO: LÓGICA Y ALERTAS
--------------------------------------

// Evaluar cumplimiento de humedad y temperatura y etiquetar cada inspección
 db.lotes.aggregate([
   { $match: { inspeccionCalidad: { $exists: true } } },
   { $project: {
       _id: 0,
       codigoLote: 1,
       aprobadoEnSistema: "$inspeccionCalidad.aprobado",
       humedad: "$inspeccionCalidad.humedad",
       temperatura: "$inspeccionCalidad.temperatura",
       resultadoCalculado: { $cond: [
         { $and: [ { $gte: ["$inspeccionCalidad.humedad", 60] }, { $lte: ["$inspeccionCalidad.humedad", 70] }, { $gte: ["$inspeccionCalidad.temperatura", 4] }, { $lte: ["$inspeccionCalidad.temperatura", 8] } ] },
         "Dentro de norma", "Fuera de norma"
       ] }
   } },
   { $match: { resultadoCalculado: "Fuera de norma" } }
 ])

// Buscar registros sospechosos: lote marcado aprobado pero con una medición fuera de rango
 db.lotes.find({
   "inspeccionCalidad.aprobado": true,
   $or: [
     { "inspeccionCalidad.humedad": { $lt: 60 } },
     { "inspeccionCalidad.humedad": { $gt: 70 } },
     { "inspeccionCalidad.temperatura": { $lt: 4 } },
     { "inspeccionCalidad.temperatura": { $gt: 8 } }
   ]
}, { codigoLote: 1, estadoActual: 1, inspeccionCalidad: 1 })

// Detectar lotes rechazados que tengan QR o despacho (posible inconsistencia)
 db.lotes.find({
   estadoActual: "Rechazado_En_Origen",
   $or: [ { qrGenerado: true }, { despacho: { $exists: true } } ]
}, { codigoLote: 1, estadoActual: 1, qrGenerado: 1, despacho: 1 })

// Detectar lotes sin inspección que ya pasaron a etapas posteriores
 db.lotes.find({

}, { codigoLote: 1, estadoActual: 1 })

       porEstado: [ { $group: { _id: "$estadoActual", cantidad: { $sum: 1 } } }, { $sort: { cantidad: -1 } } ],
       porVariedad: [ { $group: { _id: "$datosProduccion.variedad", cantidad: { $sum: 1 }, pesoTotalKg: { $sum: "$datosProduccion.pesoKg" } } }, { $sort: { pesoTotalKg: -1 } } ],
       calidad: [ { $match: { inspeccionCalidad: { $exists: true } } }, { $group: { _id: "$inspeccionCalidad.aprobado", cantidad: { $sum: 1 } } } ],
       recibidos: [ { $match: { recepcionDestino: { $exists: true } } }, { $group: { _id: null, totalCajasDanadas: { $sum: "$recepcionDestino.cajasDanadas" }, temperaturaPromedio: { $avg: "$recepcionDestino.temperaturaLlegada" } } } ]
   } }
 ])

// Historial de eventos: cantidad de eventos por tipo y actor
 db.lotes.aggregate([
   { $unwind: "$eventos" },
   { $group: { _id: { tipo: "$eventos.tipo", actor: "$eventos.actor" }, cantidad: { $sum: 1 }, lotes: { $addToSet: "$codigoLote" } } },
   { $project: { tipoEvento: "$_id.tipo", actor: "$_id.actor", cantidad: 1, lotesDistintos: { $size: "$lotes" }, _id: 0 } },
   { $sort: { cantidad: -1 } }
 ])

   { $sort: { "_id.dia": 1, "_id.tipo": 1 } }
 ])

// Ranking de fincas en base al peso total, con posición (MongoDB 5+)
